FROM python:3.12-slim

WORKDIR /app

# Instala as dependências do projeto
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copia todo o código para o container
COPY . .

# Expõe a porta padrão do Cloud Run
ENV PORT=8080

# Configura o Streamlit para rodar em modo headless na porta do Cloud Run sem abrir browser
ENV STREAMLIT_SERVER_PORT=8080
ENV STREAMLIT_SERVER_ADDRESS=0.0.0.0
ENV STREAMLIT_SERVER_HEADLESS=true

# Comando de entrada utilizando o streamlit diretamente para rodar a interface do ADK
CMD ["sh", "-c", "adk web app.py --host 0.0.0.0 --port $PORT"]
